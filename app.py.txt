def get_cartoon_avatar(name, type_resto):
    """Génère une illustration type cartoon/dessin via API DiceBear"""
    seed = urllib.parse.quote(f"{name}{type_resto}")
    # Style 'notionists' pour un look moderne et clean
    return f"https://api.dicebear.com/7.x/notionists/svg?seed={seed}&backgroundColor=ffecd2"

st.title("Marrakech Foodie Guide 🕌")
st.markdown("Découvrez les pépites culinaires de la ville ocre.")

# Filtres
col1, col2 = st.columns(2)
with col1:
    city = st.selectbox("Ville", ["Marrakech", "Paris (Bientôt)", "Londres (Bientôt)"])
with col2:
    cuisine = st.selectbox("Envie de...", ["Tout", "Marocain", "Italien", "Fusion", "Bistro"])

# Chargement des données
restaurants = get_marrakech_data(cuisine)

st.write("") # Petit espace

# Boucle d'affichage des cartes
if not restaurants:
    st.info("Aucun restaurant trouvé pour cette sélection.")
else:
    for resto in restaurants:
        # Lien Profond Google Maps (Recherche précise)
        query = urllib.parse.quote(f"{resto['name']} {resto['address']}")
        maps_url = f"https://www.google.com/maps/search/?api=1&query={query}"
        
        # Image Cartoon
        img_url = get_cartoon_avatar(resto['name'], resto['type'])
        
        # Affichage HTML
        st.markdown(f"""
        <div class="resto-card">
            <div style="display: flex; align-items: start; gap: 15px;">
                <img src="{img_url}" width="90" height="90" style="border-radius: 15px; border: 2px solid #E65100; flex-shrink: 0;">
                <div style="flex-grow: 1;">
                    <h3>{resto['name']}</h3>
                    <span class="tag">{resto['type']}</span>
                    <div style="color: #f1c40f; font-weight:bold; margin-top:5px;">
                        {'★' * int(resto['rating'])} 
                        <span style="color:#aaa; font-weight:normal;">({resto['rating']}/5)</span>
                    </div>
                </div>
            </div>
            
            <div style="margin-top: 15px; padding-top: 10px; border-top: 1px dashed #eee;">
                <div style="font-size: 0.85em; color: #888; margin-bottom: 5px;">📍 {resto['address']}</div>
                {''.join([f'<div class="review">"{avis}"</div>' for avis in resto['reviews'][:2]])}
            </div>
            
            <a href="{maps_url}" target="_blank" class="maps-btn">
                🗺️ Voir sur Google Maps
            </a>
        </div>
